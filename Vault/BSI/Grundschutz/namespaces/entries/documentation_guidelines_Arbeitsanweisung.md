  .where(p => p.stmt_documentation == "Arbeitsanweisung" &&

  .sort(g => g.key);

    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link
        ]);
        first = false;
    }
}

dv.table(
    ["Verzeichnis", "Control"],
    rows
);
