  .where(p => p.stmt_documentation == "Compliance-Verpflichtungen" &&
              p.file.etags.includes("#oscal_control"))

  .groupBy(p => p.file.folder)
  .sort(g => g.key);

let rows = [];

for (let group of pages) {
    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link
        ]);
        first = false;
    }
}

    ["Verzeichnis", "Control"],
    rows
);
