Enthält die Geschäftsprozesse, sowie deren Abhängigkeiten und Definitionen. Die Geschäftsprozesse bilden die Grundlage für das Verständnis, wie Wertschöpfung im Unternehmen entsteht und wie Informationen dabei verarbeitet werden. Ihre Beschreibung dient dazu, Abläufe transparent darzustellen, Verantwortlichkeiten zu klären und Schnittstellen sowie Abhängigkeiten sichtbar zu machen.   .where(p => p.stmt_documentation == "Geschäftsprozesse" &&
              p.file.etags.includes("#oscal_control"))
  .sort(p => p.file.name)
  .groupBy(p => p.file.folder)
  .sort(g => g.key);

let rows = [];

for (let group of pages) {
    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link
        ]);
        first = false;
    }
}

dv.table(
    ["Verzeichnis", "Control"],
    rows
);
