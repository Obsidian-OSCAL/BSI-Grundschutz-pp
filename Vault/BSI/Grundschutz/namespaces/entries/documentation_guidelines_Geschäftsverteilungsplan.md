let pages = dv.pages()
  .where(p => p.stmt_documentation == "Geschäftsverteilungsplan" &&
              p.file.etags.includes("#oscal_control"))
  .sort(p => p.file.name)
  .groupBy(p => p.file.folder)
  .sort(g => g.key);

let rows = [];

for (let group of pages) {
    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link
        ]);
        first = false;
    }
}

dv.table(
    ["Verzeichnis", "Control"],
    rows
);
