let pages = dv.pages()
  .where(p => {
    if (!p.file.etags.includes("#oscal_control") || !p.result_reference) return false;

    let refs = Array.isArray(p.result_reference)
      ? p.result_reference
      : [p.result_reference];

    return wanted.some(v => refs.includes(v));
  })
  .sort(p => p.file.name)
  .groupBy(p => p.file.folder)
  .sort(g => g.key);

let rows = [];

for (let group of pages) {
    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link
        ]);
        first = false;
    }
}

dv.table(
    ["Verzeichnis", "Control"],
