let pages = dv.pages()
  .where(p => p.topics == "Dokumentation und Kommunikation".toLowerCase())
  .sort(p => p.file.name)
  .groupBy(p => p.file.folder)
  .sort(g => g.key);

let rows = [];

for (let group of pages) {
    let first = true;
    for (let page of group.rows) {
        rows.push([
            first ? group.key : "",
            page.file.link

}
